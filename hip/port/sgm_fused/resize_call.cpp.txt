    // cheshire (step 5x): the default downscale - lanczos3 through ImageBufAlgo::resize - computed
    // directly with the same arithmetic in the same order (image/cheshireResize.hpp): the same
    // values, 18x less CPU on a 6000x3376 RGBA image, and rows in parallel unless the caller
    // already is. Anything else (a named filter, an upscale, other types) stays with OpenImageIO.
    // CHESHIRE_RESIZE_EXACT=0 restores the OpenImageIO call.
    static const bool cheshireExact = [] {
        const char* v = std::getenv("CHESHIRE_RESIZE_EXACT");
        return v == nullptr || std::string(v) != "0";
    }();
    if (cheshireExact && typeDesc == oiio::TypeDesc::FLOAT && filter.empty() && filterSize == 0.0f && outWidth <= inWidth &&
        outHeight <= inHeight && (nchannels == 1 || nchannels == 3 || nchannels == 4))
    {
        static std::once_flag once;
        std::call_once(once, [] {
            ALICEVISION_LOG_INFO("cheshire: resize: OpenImageIO's default downscale computed directly (identical values; CHESHIRE_RESIZE_EXACT=0 for OpenImageIO)");
        });
        const float* in = reinterpret_cast<const float*>(inBuffer);
        float* out = reinterpret_cast<float*>(outBuffer);
        const bool rows = !omp_in_parallel();
        if (nchannels == 4)
            image::cheshire::resizeExact<4>(in, inWidth, inHeight, out, outWidth, outHeight, rows);
        else if (nchannels == 3)
            image::cheshire::resizeExact<3>(in, inWidth, inHeight, out, outWidth, outHeight, rows);
        else
            image::cheshire::resizeExact<1>(in, inWidth, inHeight, out, outWidth, outHeight, rows);
        return;
    }

    const oiio::ImageBuf inBuf(oiio::ImageSpec(inWidth, inHeight, nchannels, typeDesc), const_cast<T*>(inBuffer));
    oiio::ImageBuf outBuf(oiio::ImageSpec(outWidth, outHeight, nchannels, typeDesc), outBuffer);
