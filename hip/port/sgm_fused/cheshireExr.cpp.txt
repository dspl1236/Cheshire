// CPU time of the calling thread, seconds (for the profile line)
double threadCpuSeconds()
{
#if defined(_WIN32)
    FILETIME c, e, k, u;
    if (!GetThreadTimes(GetCurrentThread(), &c, &e, &k, &u))
        return 0.0;
    auto toSec = [](const FILETIME& f) { return double((static_cast<unsigned long long>(f.dwHighDateTime) << 32) | f.dwLowDateTime) * 1e-7; };
    return toSec(k) + toSec(u);
#else
    timespec ts;
    if (clock_gettime(CLOCK_THREAD_CPUTIME_ID, &ts) != 0)
        return 0.0;
    return double(ts.tv_sec) + double(ts.tv_nsec) * 1e-9;
#endif
}

bool hasExrExtension(const std::string& path)
{
    const std::size_t dot = path.find_last_of('.');
    if (dot == std::string::npos)
        return false;
    std::string ext = path.substr(dot);
    std::transform(ext.begin(), ext.end(), ext.begin(), [](unsigned char c) { return static_cast<char>(std::tolower(c)); });
    return ext == ".exr";
}

// cheshire (step 6v): ZIP and ZIPS chunks inflated with libdeflate. OpenEXR before 3.2 inflates with
// zlib (the Linux bundle's 3.1); libdeflate decompresses the same stream about twice as fast, and
// decompression has one answer, so the pixels are OpenEXR's. Per chunk, as OpenEXR's zip: the zlib
// stream, then the byte predictor (each byte the previous plus itself minus 128) and the interleave
// (the first half of the bytes are the even ones), then per line the channels in the file's order

// smaller than its lines is stored uncompressed, as OpenEXR writes it. Chunks come off the file in
// order through InputFile::rawPixelData and are decoded on OpenEXR's own thread pool, in batches, so
// concurrent readers share the pool as their readPixels calls did. Default: on with OpenEXR before
// 3.2, off after (it uses libdeflate itself); CHESHIRE_EXR_DEFLATE=0|1 overrides, and
// CHESHIRE_EXR_DEFLATE_CHECK=1 also reads every such file through readPixels and keeps OpenEXR's

struct DeflateChannel
{
    int target;  // output component, or -1 when the file channel is not read
    int bytes;   // 2 (half) or 4 (float)
};

// The unpacked bytes of one chunk into out (unpacked bytes), tmp as the zlib output
bool cheshireUnzipChunk(libdeflate_decompressor* d, const char* in, std::size_t inSize, std::size_t unpacked, std::vector<unsigned char>& tmp, unsigned char* out)
{
    if (inSize >= unpacked)
    {
        if (inSize != unpacked)
            return false;
        std::memcpy(out, in, unpacked);
        return true;
    }
    tmp.resize(unpacked);
    std::size_t actual = 0;
    if (libdeflate_zlib_decompress(d, in, inSize, tmp.data(), unpacked, &actual) != LIBDEFLATE_SUCCESS || actual != unpacked)
        return false;
    unsigned char* t = tmp.data();
    for (std::size_t i = 1; i < unpacked; ++i)
        t[i] = static_cast<unsigned char>(int(t[i - 1]) + int(t[i]) - 128);
    const unsigned char* t1 = t;
    const unsigned char* t2 = t + (unpacked + 1) / 2;
    for (std::size_t i = 0; i < unpacked; ++i)
        out[i] = (i & 1) ? t2[i >> 1] : t1[i >> 1];
    return true;
}

// Chunks [k0, k1): unzip, then scatter the lines into the float image
bool cheshireDecodeChunks(const std::vector<char>& blob, const std::vector<std::size_t>& offs, int k0, int k1, int linesPerChunk, int width, int height,
                          const std::vector<DeflateChannel>& chs, std::size_t bytesPerLine, int nchannels, float* base)
{
    libdeflate_decompressor* d = libdeflate_alloc_decompressor();
    if (d == nullptr)
        return false;
    std::vector<unsigned char> tmp, unpackedBuf;
    bool good = true;
    for (int k = k0; k < k1 && good; ++k)
    {
        const int y0 = k * linesPerChunk;
        const int y1 = std::min(y0 + linesPerChunk, height);
        const std::size_t unpacked = bytesPerLine * static_cast<std::size_t>(y1 - y0);
        unpackedBuf.resize(unpacked);
        good = cheshireUnzipChunk(d, blob.data() + offs[k], offs[k + 1] - offs[k], unpacked, tmp, unpackedBuf.data());
        if (!good)
            break;
        const unsigned char* p = unpackedBuf.data();
        for (int y = y0; y < y1; ++y)
        {
            float* row = base + static_cast<std::size_t>(y) * static_cast<std::size_t>(width) * static_cast<std::size_t>(nchannels);
            for (const DeflateChannel& ch : chs)
            {
                if (ch.target < 0)
                {
                    p += static_cast<std::size_t>(ch.bytes) * static_cast<std::size_t>(width);
                    continue;
                }
                float* o = row + ch.target;
                if (ch.bytes == 2)
                {
                    for (int x = 0; x < width; ++x, p += 2, o += nchannels)
                    {
                        Imath::half h;
                        h.setBits(static_cast<unsigned short>(p[0] | (p[1] << 8)));
                        *o = static_cast<float>(h);
                    }
                }
                else
                {
                    for (int x = 0; x < width; ++x, p += 4, o += nchannels)
                        std::memcpy(o, p, 4);
                }
            }
        }
    }
    libdeflate_free_decompressor(d);
    return good;
}

class CheshireDeflateTask : public IlmThread::Task
{
  public:
    CheshireDeflateTask(IlmThread::TaskGroup* group, std::function<bool()> work, std::atomic<bool>* failed)
      : IlmThread::Task(group),
        _work(std::move(work)),
        _failed(failed)
    {}
    void execute() override
    {
        if (!_work())
            _failed->store(true);
    }

  private:
    std::function<bool()> _work;
    std::atomic<bool>* _failed;
};

// The whole image through libdeflate; false (the image then read by readPixels) on anything unexpected
bool cheshireReadExrDeflate(Imf::InputFile& file, const Imf::Header& header, int width, int height, int nchannels, const char* const* names, float* base,
                            int fileThreads)
{
    const Imf::Compression comp = header.compression();
    const int linesPerChunk = comp == Imf::ZIPS_COMPRESSION ? 1 : 16;
    std::vector<DeflateChannel> chs;
    std::size_t bytesPerPixel = 0;
    for (Imf::ChannelList::ConstIterator it = header.channels().begin(); it != header.channels().end(); ++it)
    {
        DeflateChannel ch{-1, it.channel().type == Imf::HALF ? 2 : 4};
        if (it.channel().type != Imf::HALF && it.channel().type != Imf::FLOAT)
            return false;
        for (int c = 0; c < nchannels; ++c)
            if (std::strcmp(it.name(), names[c]) == 0)
                ch.target = c;
        bytesPerPixel += static_cast<std::size_t>(ch.bytes);
        chs.push_back(ch);
    }
    const std::size_t bytesPerLine = bytesPerPixel * static_cast<std::size_t>(width);
    const int nChunks = (height + linesPerChunk - 1) / linesPerChunk;
    // the compressed chunks, in order
    std::vector<char> blob;
    std::vector<std::size_t> offs(static_cast<std::size_t>(nChunks) + 1, 0);
    for (int k = 0; k < nChunks; ++k)
    {
        const char* data = nullptr;
        int size = 0;
        file.rawPixelData(k * linesPerChunk, data, size);
        if (data == nullptr || size <= 0)
            return false;
        offs[k] = blob.size();
        blob.insert(blob.end(), data, data + size);
    }
    offs[nChunks] = blob.size();
    // decode: inline, or in batches on OpenEXR's pool like readPixels
    if (fileThreads <= 0 || nChunks < 2)
        return cheshireDecodeChunks(blob, offs, 0, nChunks, linesPerChunk, width, height, chs, bytesPerLine, nchannels, base);
    const int batch = std::max(1, 64 / linesPerChunk);
    std::atomic<bool> failed{false};
    {
        IlmThread::TaskGroup group;
        for (int k0 = 0; k0 < nChunks; k0 += batch)
        {
            const int k1 = std::min(nChunks, k0 + batch);
            IlmThread::ThreadPool::addGlobalTask(new CheshireDeflateTask(
              &group,
              [&, k0, k1] { return cheshireDecodeChunks(blob, offs, k0, k1, linesPerChunk, width, height, chs, bytesPerLine, nchannels, base); },
              &failed));
        }
    }  // the group waits for its tasks
    return !failed.load();
}

std::int32_t exrInt32(const std::vector<unsigned char>& f, std::size_t at)
{
    std::int32_t v;
    std::memcpy(&v, f.data() + at, 4);
    return v;
}

// the NUL-terminated string at p; p then points past the NUL
bool exrString(const std::vector<unsigned char>& f, std::size_t& p, std::string& s)
{
    if (p >= f.size())
        return false;
    const unsigned char* b = f.data() + p;
    const void* nul = std::memchr(b, 0, f.size() - p);
    if (nul == nullptr)
        return false;
    const std::size_t n = static_cast<const unsigned char*>(nul) - b;
    s.assign(reinterpret_cast<const char*>(b), n);
    p += n + 1;
    return true;
}

// What the deflate writer needs of the uncompressed single-part scanline EXR OpenImageIO wrote into memory
struct UncompressedExr
{
    std::size_t headerEnd = 0;      // past the header's terminating NUL: where the offset table starts
    std::size_t compressionAt = 0;  // the compression attribute's one value byte
    int ymin = 0, ymax = -1;
    std::size_t bytesPerLine = 0;
    std::vector<std::size_t> lines;  // where each line's pixel data starts
};

bool parseUncompressedExr(const std::vector<unsigned char>& f, UncompressedExr& u)
{
    if (f.size() < 8 || exrInt32(f, 0) != 20000630)
        return false;
    const std::int32_t version = exrInt32(f, 4);
    if ((version & 0xff) != 2 || (version & (0x200 | 0x800 | 0x1000)) != 0)
        return false;  // tiled, deep or multi-part
    std::size_t p = 8;
    bool haveCompression = false, haveWindow = false;
    int xmin = 0, xmax = -1;
    std::size_t bytesPerPixel = 0;
    for (;;)
    {
        if (p >= f.size())
            return false;
        if (f[p] == 0)
        {
            ++p;
            break;
        }
        std::string name, type;
        if (!exrString(f, p, name) || !exrString(f, p, type) || p + 4 > f.size())
            return false;
        const std::int32_t size = exrInt32(f, p);
        p += 4;
        if (size < 0 || p + std::size_t(size) > f.size())
            return false;
        if (name == "compression")
        {
            if (size != 1 || f[p] != 0)
                return false;  // not the uncompressed file asked for
            u.compressionAt = p;
            haveCompression = true;
        }
        else if (name == "dataWindow")
        {
            if (size != 16)
                return false;
            xmin = exrInt32(f, p);
            u.ymin = exrInt32(f, p + 4);
            xmax = exrInt32(f, p + 8);
            u.ymax = exrInt32(f, p + 12);
            haveWindow = true;
        }
        else if (name == "lineOrder")
        {
            if (size != 1 || f[p] != 0)
                return false;  // increasing y only
        }
        else if (name == "type")
        {
            if (std::string(reinterpret_cast<const char*>(f.data() + p), std::size_t(size)).rfind("scanlineimage", 0) != 0)
                return false;
        }
        else if (name == "chunkCount" || name == "tiles")
            return false;  // an attribute the chunk layout would have to follow
        else if (name == "channels")
        {
            std::size_t q = p;
            const std::size_t end = p + std::size_t(size);
            while (q < end && f[q] != 0)
            {
                std::string channel;
                if (!exrString(f, q, channel) || q + 16 > end)
                    return false;
                const std::int32_t pixelType = exrInt32(f, q);
                if (exrInt32(f, q + 8) != 1 || exrInt32(f, q + 12) != 1 || pixelType < 0 || pixelType > 2)
                    return false;
                bytesPerPixel += pixelType == 1 ? 2 : 4;  // HALF; UINT and FLOAT are 4
                q += 16;
            }
        }
        p += std::size_t(size);
    }
    if (!haveCompression || !haveWindow || bytesPerPixel == 0 || xmax < xmin || u.ymax < u.ymin)
        return false;
    u.headerEnd = p;
    u.bytesPerLine = bytesPerPixel * std::size_t(xmax - xmin + 1);
    const std::size_t height = std::size_t(u.ymax - u.ymin) + 1;
    if (p + 8 * height > f.size())
        return false;
    u.lines.resize(height);
    for (std::size_t i = 0; i < height; ++i)
    {
        std::uint64_t at;
        std::memcpy(&at, f.data() + p + 8 * i, 8);
        if (at > f.size() || f.size() - at < 8 + u.bytesPerLine || exrInt32(f, std::size_t(at)) != u.ymin + std::int32_t(i) ||
            exrInt32(f, std::size_t(at) + 4) != std::int32_t(u.bytesPerLine))
            return false;
        u.lines[i] = std::size_t(at) + 8;
    }
    return true;
}

// buf written to path as OpenImageIO writes it with ZIP (16 lines a chunk) or ZIPS (one line) at level, the chunks
// deflated by libdeflate; false (the file removed) on anything unexpected
bool writeExrDeflate(const OIIO::ImageBuf& buf, const std::string& path, bool zips, int level)
{
    // OpenImageIO's uncompressed file, in memory
    std::vector<unsigned char> mem;
    mem.reserve(std::size_t(buf.spec().image_bytes()) + 16 * std::size_t(std::max(0, buf.spec().height)) + (std::size_t(1) << 16));
    {
        const void* pixels = buf.localpixels();
        auto out = OIIO::ImageOutput::create("openexr");
        if (pixels == nullptr || !out || !out->supports("ioproxy"))
            return false;
        OIIO::Filesystem::IOVecOutput proxy(mem);
        OIIO::ImageSpec spec = buf.spec();
        spec.attribute("compression", "none");
        if (!out->set_ioproxy(&proxy) || !out->open(path, spec))
            return false;
        if (!out->write_image(spec.format, pixels, buf.pixel_stride(), buf.scanline_stride(), buf.z_stride()) || !out->close())
            return false;
    }
    UncompressedExr u;
    if (!parseUncompressedExr(mem, u) || u.bytesPerLine % 2 != 0)
        return false;

    const std::size_t linesPerChunk = zips ? 1 : 16;
    const std::size_t height = u.lines.size();
    const std::size_t nChunks = (height + linesPerChunk - 1) / linesPerChunk;
    libdeflate_compressor* compressor = libdeflate_alloc_compressor(level);
    if (compressor == nullptr)
        return false;
    FILE* fp = OIIO::Filesystem::fopen(path, "wb");
    if (fp == nullptr)
    {
        libdeflate_free_compressor(compressor);
        return false;
    }
    bool ok = true;
    std::uint64_t pos = 0;
    auto put = [&](const void* data, std::size_t n) {
        if (ok && n > 0 && std::fwrite(data, 1, n, fp) != n)
            ok = false;
        pos += n;
    };
    // the header with compression ZIP (3) or ZIPS (2), then the offset table, filled in at the end
    mem[u.compressionAt] = zips ? 2 : 3;
    put(mem.data(), u.headerEnd);
    const std::uint64_t tableAt = pos;
    std::vector<std::uint64_t> table(nChunks, 0);
    put(table.data(), 8 * nChunks);
    std::vector<unsigned char> inter, pred, packed;
    for (std::size_t k = 0; k < nChunks && ok; ++k)
    {
        const std::size_t first = k * linesPerChunk;
        const std::size_t count = std::min(linesPerChunk, height - first);
        const std::size_t n = count * u.bytesPerLine;
        // OpenEXR's zip: the interleave (the even bytes of the chunk's lines, then the odd ones), straight from the
        // lines - their length is even, so each continues the split - then the predictor
        inter.resize(n);
        unsigned char* t1 = inter.data();
        unsigned char* t2 = inter.data() + n / 2;
        for (std::size_t i = 0; i < count; ++i)
        {
            const unsigned char* s = mem.data() + u.lines[first + i];
            for (std::size_t b = 0; b < u.bytesPerLine; b += 2)
            {
                *t1++ = s[b];
                *t2++ = s[b + 1];
            }
        }
        pred.resize(n);
        pred[0] = inter[0];
        for (std::size_t i = 1; i < n; ++i)
            pred[i] = static_cast<unsigned char>(int(inter[i]) - int(inter[i - 1]) + (128 + 256));
        // the zlib stream only when it is smaller than the lines, as OpenEXR decides
        packed.resize(n);
        const std::size_t z = libdeflate_zlib_compress(compressor, pred.data(), n, packed.data(), n - 1);
        table[k] = pos;
        const std::int32_t y = u.ymin + static_cast<std::int32_t>(first);
        const std::int32_t size = static_cast<std::int32_t>(z > 0 ? z : n);
        put(&y, 4);
        put(&size, 4);
        if (z > 0)
            put(packed.data(), z);
        else
            for (std::size_t i = 0; i < count; ++i)
                put(mem.data() + u.lines[first + i], u.bytesPerLine);
    }
    libdeflate_free_compressor(compressor);
    if (ok && std::fseek(fp, static_cast<long>(tableAt), SEEK_SET) == 0)
        put(table.data(), 8 * nChunks);
    else
        ok = false;
    ok = std::fclose(fp) == 0 && ok;
    if (!ok)
        OIIO::Filesystem::remove(path);
    return ok;
}

// The file at path read through OpenEXR (its zlib inflate): are its pixels buf's, bit for bit?
bool exrReadsBackAs(const std::string& path, const OIIO::ImageBuf& buf)
{
    const OIIO::ImageSpec& spec = buf.spec();
    if (spec.format != OIIO::TypeDesc::HALF && spec.format != OIIO::TypeDesc::FLOAT)
        return false;
    Imf::InputFile file(path.c_str());
    const Imath::Box2i dw = file.header().dataWindow();
    const int width = dw.max.x - dw.min.x + 1, height = dw.max.y - dw.min.y + 1;
    if (width != spec.width || height != spec.height)
        return false;
    const std::size_t elem = spec.format.size();
    const std::size_t xs = elem * std::size_t(spec.nchannels), ys = xs * std::size_t(width);
    std::vector<char> back(ys * std::size_t(height));
    Imf::FrameBuffer fb;
    char* origin = back.data() - (std::ptrdiff_t(dw.min.x) * std::ptrdiff_t(xs) + std::ptrdiff_t(dw.min.y) * std::ptrdiff_t(ys));
    for (int c = 0; c < spec.nchannels; ++c)
        fb.insert(spec.channelnames[c].c_str(), Imf::Slice(elem == 2 ? Imf::HALF : Imf::FLOAT, origin + std::size_t(c) * elem, xs, ys));
    file.setFrameBuffer(fb);
    file.readPixels(dw.min.y, dw.max.y);
    const char* src = static_cast<const char*>(buf.localpixels());
    if (src == nullptr || buf.pixel_stride() != OIIO::stride_t(xs))
        return false;
    for (int y = 0; y < height; ++y)
        if (std::memcmp(back.data() + std::size_t(y) * ys, src + std::ptrdiff_t(y) * buf.scanline_stride(), ys) != 0)
            return false;
    return true;
}
#endif

ConcurrentLoadScope::ConcurrentLoadScope() { ++tlConcurrentLoad; }
ConcurrentLoadScope::~ConcurrentLoadScope() { --tlConcurrentLoad; }
bool inConcurrentLoad() { return tlConcurrentLoad > 0; }

ExrDeflateWriteScope::ExrDeflateWriteScope() { ++tlExrDeflateWrite; }
ExrDeflateWriteScope::~ExrDeflateWriteScope() { --tlExrDeflateWrite; }
bool inExrDeflateWrite() { return tlExrDeflateWrite > 0; }

bool cheshireWriteExrDeflate(const OIIO::ImageBuf& buf, const std::string& path, const std::string& compression)
{
#if defined(CHESHIRE_HAVE_DEFLATE)
    if (!cheshireExrDeflateWriteEnabled())
        return false;
    // "zip:<level>" or "zips:<level>", the level given (without one OpenEXR picks its own)
    const std::size_t colon = compression.find(':');
    const std::string method = compression.substr(0, colon);
    if ((method != "zip" && method != "zips") || colon == std::string::npos)
        return false;
    const int level = std::atoi(compression.c_str() + colon + 1);
    if (level < 1 || level > 9)
        return false;
    try
    {
        if (!writeExrDeflate(buf, path, method == "zips", level))
            return false;
        ++g_cheshireDeflateWriteStats.written;
        if (!g_cheshireDeflateWriteStats.announced.exchange(true))
            ALICEVISION_LOG_INFO("cheshire: EXR ZIP/ZIPS chunks deflated with libdeflate on write (OpenEXR "
                                 << OPENEXR_VERSION_MAJOR << "." << OPENEXR_VERSION_MINOR
                                 << "; CHESHIRE_EXR_DEFLATE_WRITE=0 for OpenEXR's own, CHESHIRE_EXR_DEFLATE_WRITE_CHECK=1 to read every file back)");
        if (cheshireExrDeflateWriteCheck())
        {
            ++g_cheshireDeflateWriteStats.checked;
            if (!exrReadsBackAs(path, buf))
            {
                ALICEVISION_LOG_WARNING("cheshire: EXR deflate write check: '" << path << "' does not read back as written; OpenImageIO writes it");
                OIIO::Filesystem::remove(path);
                return false;
            }
            ++g_cheshireDeflateWriteStats.identical;
        }
        return true;
    }
    catch (const std::exception& e)
    {
        ALICEVISION_LOG_DEBUG("[IO] cheshire: deflate write of '" << path << "' not taken (" << e.what() << "), OpenImageIO path");
        OIIO::Filesystem::remove(path);
        return false;
    }
#else
    (void)buf;
    (void)path;
    (void)compression;
    return false;
#endif
}

bool cheshireReadExr(const std::string& path, int nchannels, bool requireLinearMetadata, const std::function<float*(int, int)>& allocate)
{
    if (!cheshireExrDirectEnabled() || (nchannels != 1 && nchannels != 3 && nchannels != 4) || !hasExrExtension(path))
        return false;
    try
    {
        static std::once_flag once;
        std::call_once(once, [] {
            int threads = static_cast<int>(std::max(1u, std::thread::hardware_concurrency()));
            threads = static_cast<int>(std::max(0LL, ::cheshire::env::integer("CHESHIRE_EXR_THREADS", threads)));
            Imf::setGlobalThreadCount(threads);
            ALICEVISION_LOG_INFO("cheshire: EXR files read through OpenEXR directly (one-channel, RGB and RGBA half or float, linear; " << threads
                                 << " pool threads for a lone read, inline inside a parallel loop or a read-ahead thread; CHESHIRE_EXR_DIRECT=0 for OpenImageIO)");
        });
        // A lone read (texturing, one image at a time) spreads its scanlines over the pool. Inside a
        // parallel loop (the depth-map prefetch, a dozen files at once) every file decodes in its own
        // calling thread instead: with all of them queueing 3376 single-scanline tasks into one pool,
        // twenty files took as long as twenty sequential reads.
        // ... and so does a read-ahead thread of the image cache (ConcurrentLoadScope, step 6c): the
        // texturing node reads 4 cameras ahead through std::async, outside any OpenMP region.
        const int fileThreads = (omp_in_parallel() || inConcurrentLoad()) ? 0 : Imf::globalThreadCount();
        const bool profile = cheshireExrProfile();
        const auto tOpen0 = std::chrono::steady_clock::now();
        const double cpu0 = profile ? threadCpuSeconds() : 0.0;
        Imf::InputFile file(path.c_str(), fileThreads);
        const auto tOpen1 = std::chrono::steady_clock::now();
        const Imf::Header& header = file.header();
        if (requireLinearMetadata)
        {
            const Imf::StringAttribute* cs = header.findTypedAttribute<Imf::StringAttribute>("AliceVision:ColorSpace");
            if (cs != nullptr && cs->value() != "linear")
                return false;
        }
        const Imath::Box2i dw = header.dataWindow();
        if (dw != header.displayWindow() || dw.min.x != 0 || dw.min.y != 0)
            return false;
        const int width = dw.max.x + 1;
        const int height = dw.max.y + 1;
        const Imf::ChannelList& channels = header.channels();
        int nbFileChannels = 0;
        for (Imf::ChannelList::ConstIterator it = channels.begin(); it != channels.end(); ++it)
            ++nbFileChannels;
        const char* names[4] = {"R", "G", "B", "A"};
        if (nchannels == 1)
        {
            // one channel requested: only a one-channel file (a depth or similarity map, channel "Y"),
            // which OpenImageIO reads as its channel 0; a colour file would mean a luminance
            // conversion, and that stays with OpenImageIO
            if (nbFileChannels != 1)
                return false;
            const Imf::Channel& channel = channels.begin().channel();
            if ((channel.type != Imf::HALF && channel.type != Imf::FLOAT) || channel.xSampling != 1 || channel.ySampling != 1)
                return false;
            names[0] = channels.begin().name();
        }
        else
        {
            int present = 0;
            for (int c = 0; c < 4; ++c)
            {
                const Imf::Channel* channel = channels.findChannel(names[c]);
                if (channel == nullptr)
                    continue;
                if ((channel->type != Imf::HALF && channel->type != Imf::FLOAT) || channel->xSampling != 1 || channel->ySampling != 1)
                    return false;
                present |= 1 << c;
            }
            if ((present & 7) != 7 || nbFileChannels != ((present & 8) ? 4 : 3) || (nchannels == 4 && !(present & 8)))
                return false;  // only R, G, B and optionally A; an alpha requested must exist
        }

        float* base = allocate(width, height);
        const std::size_t xStride = sizeof(float) * static_cast<std::size_t>(nchannels);
        const std::size_t yStride = xStride * static_cast<std::size_t>(width);
        bool viaDeflate = false;
#if defined(CHESHIRE_HAVE_DEFLATE)
        {
            const Imf::Compression compression = header.compression();
            if (!header.hasTileDescription() && (compression == Imf::ZIPS_COMPRESSION || compression == Imf::ZIP_COMPRESSION) && cheshireExrDeflateEnabled()
                && cheshireReadExrDeflate(file, header, width, height, nchannels, names, base, fileThreads))
            {
                viaDeflate = true;
                if (cheshireExrDeflateCheck())
                {
                    // OpenEXR's own pixels beside these, bit for bit; OpenEXR's are kept on a difference
                    std::vector<float> ref(static_cast<std::size_t>(width) * static_cast<std::size_t>(height) * static_cast<std::size_t>(nchannels));
                    Imf::FrameBuffer refBuffer;
                    for (int c = 0; c < nchannels; ++c)
                        refBuffer.insert(names[c], Imf::Slice(Imf::FLOAT, reinterpret_cast<char*>(ref.data() + c), xStride, yStride));
                    file.setFrameBuffer(refBuffer);
                    file.readPixels(dw.min.y, dw.max.y);
                    const bool same = std::memcmp(ref.data(), base, ref.size() * sizeof(float)) == 0;
                    ++g_cheshireDeflateStats.checked;
                    if (same)
                        ++g_cheshireDeflateStats.identical;
                    else
                    {
                        std::memcpy(base, ref.data(), ref.size() * sizeof(float));
                        ALICEVISION_LOG_WARNING("cheshire: EXR deflate check: " << path << " differs from OpenEXR's readPixels; OpenEXR's pixels kept");
                    }
                }
            }
        }
#endif
        Imf::FrameBuffer frameBuffer;
        for (int c = 0; c < nchannels; ++c)
            frameBuffer.insert(names[c], Imf::Slice(Imf::FLOAT, reinterpret_cast<char*>(base + c), xStride, yStride));
        if (viaDeflate)
        {
        }
        else if (header.hasTileDescription())
        {
            file.setFrameBuffer(frameBuffer);
            file.readPixels(dw.min.y, dw.max.y);
        }
        else
        {
            file.setFrameBuffer(frameBuffer);
            // In blocks of scanlines. readPixels allocates its packed and unpacked chunk buffers
            // for the whole requested range up front - 235 MB for one of these files, freed after -
            // and a dozen threads page-faulting fresh temporaries at once serialise in the kernel:
            // one batch of 16 files took 100 s of CPU for 11 s of wall, no less wall than a single
            // thread. Small ranges reuse the same freed blocks and the threads decode in parallel.
            constexpr int kLinesPerCall = 64;
            for (int y = dw.min.y; y <= dw.max.y; y += kLinesPerCall)
                file.readPixels(y, std::min(y + kLinesPerCall - 1, dw.max.y));
        }
        if (profile)
        {
            const auto tEnd = std::chrono::steady_clock::now();
            ALICEVISION_LOG_INFO("cheshire: EXR profile: " << path.substr(path.find_last_of("/\\\\") + 1) << " open "
                                                             << std::chrono::duration<double>(tOpen1 - tOpen0).count() << " s, read "
                                                             << std::chrono::duration<double>(tEnd - tOpen1).count() << " s, thread cpu "
                                                             << (threadCpuSeconds() - cpu0) << " s, file threads " << fileThreads << ", in parallel "
                                                             << (omp_in_parallel() ? 1 : 0) << (viaDeflate ? ", libdeflate" : ""));
        }
        return true;
    }
    catch (const std::exception& e)
    {
        ALICEVISION_LOG_DEBUG("[IO] cheshire: direct EXR read of '" << path << "' not taken (" << e.what() << "), OpenImageIO path");
        return false;
    }
}
