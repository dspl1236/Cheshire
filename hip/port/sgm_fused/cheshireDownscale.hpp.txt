/// cheshire (step 6d): CHESHIRE_DEPTHMAP_DEVICE_DOWNSCALE=0 keeps the process downscale on the host
inline bool cheshireDeviceDownscaleEnabled()
{
    static const bool on = [] {
        // on by default since its gate passed (byte-identical depth maps on mini6 and the 884-view
        // chunk); CHESHIRE_DEPTHMAP_DEVICE_DOWNSCALE_CHECK=1 compares every image with the host path
        return ::cheshire::env::flag("CHESHIRE_DEPTHMAP_DEVICE_DOWNSCALE", true);
    }();
    return on;
}

/**
 * @brief cheshire (step 6d): the depth-map node's process downscale on the device, and the conversion
 * DeviceCache::addMipmapImage does on the host after it.
 *
 * The host path is mvsUtils::loadImage's imageAlgo::resizeImage (OpenImageIO's default lanczos3
 * downscale, computed exactly by image/cheshireResize.hpp), then half(value * 255) per channel. Here
 * the full-resolution float RGBA image is uploaded, the same tap tables are computed on the host by
 * the same code and applied on the device in the same order with a multiply and an add per tap and no
 * contraction, and the same conversion is applied: the output is the host path's, bit for bit.
 *
 * @param[in] in_hostRGBA the full-resolution image, row-major float RGBA
 * @param[in] inWidth, inHeight its size
 * @param[out] out_dmp the downscaled CudaRGBA image, sized inWidth / downscale x inHeight / downscale
 */
void cuda_cheshireDownscaleToCudaRGBA(const float* in_hostRGBA, int inWidth, int inHeight, CudaDeviceMemoryPitched<CudaRGBA, 2>& out_dmp);

/**
 * @brief cheshire (step 6m): the same downscale and conversion from a full-resolution float RGBA image
 * already on the device (in_d, inPitch bytes per row), queued on the given stream and waited for.
 */
void cuda_cheshireDownscaleDeviceToCudaRGBA(const float4* in_d,
                                            std::size_t inPitch,
                                            int inWidth,
                                            int inHeight,
                                            CudaDeviceMemoryPitched<CudaRGBA, 2>& out_dmp,
                                            cudaStream_t stream);

}  // namespace depthMap
}  // namespace aliceVision
