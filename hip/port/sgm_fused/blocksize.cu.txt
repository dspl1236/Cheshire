        dim3 recommendedBlock(32, divUp(recommendedBlockSize, 32), 1);
#ifdef CHESHIRE_HIP
        // cheshire: hipOccupancyMaxPotentialBlockSize tends to answer 1024 on RDNA, i.e. 32x32
        // blocks for register-heavy kernels. CHESHIRE_BLOCK_Y=<n> overrides the block height.
        static int overrideY = -2;
        if (overrideY == -2)
        {
            overrideY = static_cast<int>(::cheshire::env::integer("CHESHIRE_BLOCK_Y", 0));
            if (::cheshire::env::flag("CHESHIRE_PROFILE_SGM"))
                std::fprintf(stderr, "[cheshire-prof] occupancy block for kernels: 32x%d (min grid %d), CHESHIRE_BLOCK_Y=%d\n",
                             int(divUp(recommendedBlockSize, 32)), recommendedMinGridSize, overrideY);
        }
        if (overrideY > 0)
            recommendedBlock.y = overrideY;
#endif
        return recommendedBlock;
    }
