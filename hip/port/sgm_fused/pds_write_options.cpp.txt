// cheshire (step 5y): the EXR compression of the undistorted images. Upstream leaves the writer on
// its default, ZIPS - one zlib stream per scanline, 3376 of them for a 6000x3376 image - and every
// later node pays to inflate them: the depth-map node loads each image about 3.5 times over a job,
// texturing once per atlas sheet. ZIP compresses 16 scanlines per block: fewer streams, a slightly
// better ratio, and a cheaper read and write, with the same pixels. CHESHIRE_PDS_EXR_COMPRESSION
// names another method (none, rle, zip, zips, piz, pxr24, b44, b44a, dwaa, dwab) - the lossy ones
// change the values and are for experiments only.
static image::ImageWriteOptions cheshirePdsWriteOptions()
{
    static const image::ImageWriteOptions options = [] {
        const std::string method = ::cheshire::env::text("CHESHIRE_PDS_EXR_COMPRESSION", "zip:1");
        image::ImageWriteOptions o;
        // cheshire (step 6f): "method:level". ZIP at level 1 writes a 6000x3376 half RGBA image in 0.95 s
        // against 1.45 s at OpenEXR's default level 4, for a file 1.7 % larger, the same read time and
        // the same pixels (build/exrbench/exrwrite.cpp).
        const std::size_t colon = method.find(':');
        o.exrCompressionMethod(image::EImageExrCompression_stringToEnum(method.substr(0, colon)));
        if (colon != std::string::npos)
            o.exrCompressionLevel(std::atoi(method.c_str() + colon + 1));
        ALICEVISION_LOG_INFO("cheshire: undistorted images written with EXR compression '" << method
                             << "' (CHESHIRE_PDS_EXR_COMPRESSION=method[:level] to change; upstream's default is zips at level 4)");
        return o;
    }();
    return options;
}
