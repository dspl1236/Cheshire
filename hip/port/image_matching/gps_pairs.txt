// cheshire (step 6w): pairs from GPS positions, opt-in (CHESHIRE_GPS_PAIRING_RADIUS=<metres>).
// Above minNbImages Meshroom's ImageMatching takes each photo's partners from the vocabulary tree.
// On a 444-photo drone survey of uniform ground those partners were a median 214 m apart, and SfM
// placed 49 photos; the real pairs of the exhaustive rerun sit a median 29 m apart, 99 % within
// 56 m (docs/04, "A 444-photo drone survey"). With a radius, every pair of views that both carry GPS
// comes from their distance (the EXIF position in Cartesian metres, WGS84ToCartesian), and the
// method's pairs between two GPS views are dropped, or kept too with CHESHIRE_GPS_PAIRING_UNION=1.
// Pairs with a view that has no GPS stay as the method proposed them, so a mixed set still pairs.
// A pair is added once, in either orientation.
void cheshireGpsPairs(const sfmData::SfMData& sfmData, double radius, bool keepMethodPairs, EImageMatchingMethod method, OrderedPairList& pairs)
{
    std::vector<ImageID> ids;
    std::vector<Vec3> positions;
    for (const auto& v : sfmData.getViews())
    {
        const sfmData::ImageInfo& image = v.second->getImage();
        // Latitude and longitude with their references are required; the altitude is optional, and so
        // is GPSAltitudeRef, which EXIF defines as 0 (above sea level) when absent. Upstream's
        // hasGpsMetadata() requires all six tags, and so dropped every photo of a camera that omits
        // the reference: the Sony DSC-WX220 of OpenDroneMap's zoo survey, 0 of 524 views found.
        using sfmData::GPSExifTags;
        if (!image.hasMetadata({GPSExifTags::latitude()}) || !image.hasMetadata({GPSExifTags::latitudeRef()}) ||
            !image.hasMetadata({GPSExifTags::longitude()}) || !image.hasMetadata({GPSExifTags::longitudeRef()}))
            continue;
        try
        {
            const auto& meta = image.getMetadata();
            const double lat = parseGPSFromString(meta.at(GPSExifTags::latitude()), meta.at(GPSExifTags::latitudeRef()));
            const double lon = parseGPSFromString(meta.at(GPSExifTags::longitude()), meta.at(GPSExifTags::longitudeRef()));
            double alt = 0.0;
            if (image.hasMetadata({GPSExifTags::altitude()}))
                alt = parseAltitudeFromString(meta.at(GPSExifTags::altitude()),
                                              image.hasMetadata({GPSExifTags::altitudeRef()}) ? meta.at(GPSExifTags::altitudeRef()) : std::string("0"));
            const Vec3 p = WGS84ToCartesian(Vec3(lat, lon, alt));
            if (p.allFinite())
            {
                ids.push_back(v.first);
                positions.push_back(p);
            }
        }
        catch (const std::exception&)
        {
        }
    }
    const std::size_t nbViews = sfmData.getViews().size();
    if (ids.size() < 2)
    {
        ALICEVISION_LOG_WARNING("cheshire: GPS pairing: " << ids.size() << " of " << nbViews << " views carry GPS, the "
                                                         << EImageMatchingMethod_enumToString(method) << " pairs are kept");
        return;
    }
    const std::set<ImageID> gpsViews(ids.begin(), ids.end());
    OrderedPairList out;
    std::size_t methodBetweenGps = 0, methodWithoutGps = 0;
    for (const auto& a : pairs)
        for (const ImageID b : a.second)
        {
            const bool bothGps = gpsViews.count(a.first) != 0 && gpsViews.count(b) != 0;
            (bothGps ? methodBetweenGps : methodWithoutGps) += 1;
            if (!bothGps || keepMethodPairs)
                out[a.first].insert(b);
        }
    const double r2 = radius * radius;
    std::vector<double> distances;
    std::size_t added = 0;
    for (std::size_t i = 0; i < ids.size(); ++i)
        for (std::size_t j = i + 1; j < ids.size(); ++j)
        {
            const double d2 = (positions[i] - positions[j]).squaredNorm();
            if (d2 > r2)
                continue;
            distances.push_back(std::sqrt(d2));
            const ImageID a = std::min(ids[i], ids[j]), b = std::max(ids[i], ids[j]);
            const auto reverse = out.find(b);
            if (reverse != out.end() && reverse->second.count(a) != 0)
                continue;  // the method already has this pair the other way round
            if (out[a].insert(b).second)
                ++added;
        }
    double median = 0.0;
    if (!distances.empty())
    {
        std::nth_element(distances.begin(), distances.begin() + distances.size() / 2, distances.end());
        median = distances[distances.size() / 2];
    }
    ALICEVISION_LOG_INFO("cheshire: GPS pairing: radius " << radius << " m, " << ids.size() << " of " << nbViews << " views carry GPS; "
                                                        << distances.size() << " pairs within the radius (median " << median << " m), " << added
                                                        << " of them added; the " << EImageMatchingMethod_enumToString(method) << " method proposed "
                                                        << methodBetweenGps << " pairs between GPS views (" << (keepMethodPairs ? "kept" : "dropped")
                                                        << ") and " << methodWithoutGps << " with a view without GPS (kept)");
    pairs.swap(out);
}
