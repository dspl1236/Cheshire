    /// capacities of the edges from the source side (empty cells and the source) to the sink side
    /// (full cells and the sink). CHESHIRE_MAXFLOW_CHECK evaluates the GPU labelling and this graph's
    /// own on the same edges: a minimum cut need not be unique, and two labellings that disagree on
    /// cells a cut leaves free are equally good when their values are equal.
    double cheshireCutValue(const std::vector<bool>& full) const
    {
        const auto sinkSide = [&](std::size_t n) {
            return n == std::size_t(_T) || (n != std::size_t(_S) && n < full.size() && full[n]);
        };
        double v = 0;
        typename boost::graph_traits<Graph>::vertex_iterator vi, ve;
        for (boost::tie(vi, ve) = boost::vertices(_graph); vi != ve; ++vi)
        {
            if (sinkSide(std::size_t(*vi)))
                continue;
            typename boost::graph_traits<Graph>::out_edge_iterator ei, ee;
            for (boost::tie(ei, ee) = boost::out_edges(*vi, _graph); ei != ee; ++ei)
                if (sinkSide(std::size_t(boost::target(*ei, _graph))))
                    v += double(_graph[*ei].capacity);
        }
        return v;
    }
