// cheshire (step 6i): the arrays a split appends to grew by a fixed 1000 or 3000 entries, so the
// first cleaning pass of a large mesh reallocated and copied them every few hundred splits (the
// 24-million-entry edge array among them). They now grow by an eighth of their capacity at least:
// the same contents, a few reallocations.
template<class T>
static void cheshireReserveAddIfNeeded(StaticVector<T>& v, int nplanned, int ntoallocated)
{
    if (v.size() + nplanned > v.capacity())
        v.reserveAdd(std::max<int>(nplanned + ntoallocated, static_cast<int>(v.capacity() / 8)));
}

struct CheshireCleanState
{

    StaticVector<StaticVector<int>> ptsNeighPtsOrdered;
    StaticVectorBool ptsBoundary;

    StaticVectorBool edgesNeigTrisAlive;
    StaticVector<Voxel> edgesNeigTris;
    StaticVector<Voxel> edgesXStat;
    StaticVector<Voxel> edgesXYStat;
    int nPtsInit = 0;

    void capture(MeshClean& m)
    {
        pts = m.pts; tris = m.tris; colors = m.colors();
        ptsNeighTrisSortedAsc = m.ptsNeighTrisSortedAsc; ptsNeighPtsOrdered = m.ptsNeighPtsOrdered;
        ptsBoundary = m.ptsBoundary; newPtsOldPtId = m.newPtsOldPtId;
        edgesNeigTrisAlive = m.edgesNeigTrisAlive; edgesNeigTris = m.edgesNeigTris;
        edgesXStat = m.edgesXStat; edgesXYStat = m.edgesXYStat; nPtsInit = m.nPtsInit;
    }
    void restore(MeshClean& m) const
    {
        m.pts = pts; m.tris = tris; m.colors() = colors;
        m.ptsNeighTrisSortedAsc = ptsNeighTrisSortedAsc; m.ptsNeighPtsOrdered = ptsNeighPtsOrdered;
        m.ptsBoundary = ptsBoundary; m.newPtsOldPtId = newPtsOldPtId;
        m.edgesNeigTrisAlive = edgesNeigTrisAlive; m.edgesNeigTris = edgesNeigTris;
        m.edgesXStat = edgesXStat; m.edgesXYStat = edgesXYStat; m.nPtsInit = nPtsInit;
    }
};

template<class T, class Eq>
std::size_t cheshireCountDiff(const StaticVector<T>& a, const StaticVector<T>& b, Eq eq)
{
    std::size_t n = a.size() > b.size() ? a.size() - b.size() : b.size() - a.size();
    const int m = std::min(a.size(), b.size());
    for (int i = 0; i < m; ++i)
        n += eq(a[i], b[i]) ? 0 : 1;
    return n;
}

// names and counts of the structures that differ; empty when the two states are the same
std::string cheshireCompare(const CheshireCleanState& a, const CheshireCleanState& b)
{
    std::string out;
    auto note = [&](const char* name, std::size_t n) {
        if (n != 0)
            out += std::string(out.empty() ? "" : ", ") + name + " " + std::to_string(n);
    };
    note("pts", cheshireCountDiff(a.pts, b.pts, [](const Point3d& x, const Point3d& y) {
             return std::memcmp(&x.x, &y.x, sizeof(double)) == 0 && std::memcmp(&x.y, &y.y, sizeof(double)) == 0 &&
                    std::memcmp(&x.z, &y.z, sizeof(double)) == 0; }));
    note("tris", cheshireCountDiff(a.tris, b.tris, [](const Mesh::triangle& x, const Mesh::triangle& y) {
             return x.v[0] == y.v[0] && x.v[1] == y.v[1] && x.v[2] == y.v[2] && x.alive == y.alive; }));
    {
        std::size_t n = a.colors.size() > b.colors.size() ? a.colors.size() - b.colors.size() : b.colors.size() - a.colors.size();
        for (std::size_t i = 0; i < std::min(a.colors.size(), b.colors.size()); ++i)
            n += (a.colors[i].r == b.colors[i].r && a.colors[i].g == b.colors[i].g && a.colors[i].b == b.colors[i].b) ? 0 : 1;
        note("colors", n);
    }
    auto sameList = [](const StaticVector<int>& x, const StaticVector<int>& y) { return x.getData() == y.getData(); };
    note("ptsNeighTrisSortedAsc", cheshireCountDiff(a.ptsNeighTrisSortedAsc, b.ptsNeighTrisSortedAsc, sameList));
    note("ptsNeighPtsOrdered", cheshireCountDiff(a.ptsNeighPtsOrdered, b.ptsNeighPtsOrdered, sameList));
    note("ptsBoundary", cheshireCountDiff(a.ptsBoundary, b.ptsBoundary, [](char x, char y) { return x == y; }));
    note("newPtsOldPtId", cheshireCountDiff(a.newPtsOldPtId, b.newPtsOldPtId, [](int x, int y) { return x == y; }));
    note("edgesNeigTrisAlive", cheshireCountDiff(a.edgesNeigTrisAlive, b.edgesNeigTrisAlive, [](char x, char y) { return x == y; }));
    auto sameVoxel = [](const Voxel& x, const Voxel& y) { return x.x == y.x && x.y == y.y && x.z == y.z; };
    note("edgesNeigTris", cheshireCountDiff(a.edgesNeigTris, b.edgesNeigTris, sameVoxel));
    note("edgesXStat", cheshireCountDiff(a.edgesXStat, b.edgesXStat, sameVoxel));
    note("edgesXYStat", cheshireCountDiff(a.edgesXYStat, b.edgesXYStat, sameVoxel));
    note("nPtsInit", a.nPtsInit == b.nPtsInit ? 0 : 1);
    return out;
}
