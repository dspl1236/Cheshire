// PrepareDenseScene's undistortion on the device (hip/port/cheshireremap): the remappers, their bound and the
// switches. A remapper holds a source and a destination RGBA float image on the device and one in pinned
// staging (288 MB each at 4896x3672) plus the map. There are at most CHESHIRE_UNDISTORT_DEVICE_REMAPPERS of them
// (default: the hardware threads, up to 4), and no more than 60 % of the device memory free at the first image
// holds. An image thread that finds none free runs the CPU loop instead of waiting. The device's side of an image
// is mostly copies - through pinned memory and over the bus - and on the twelve-thread RX 9070 box those cost more
// than that CPU's loop (docs/04, 0.3.8), so the device takes the undistortion by default only on a host of four
// hardware threads or fewer, where the loop is the costly side. CHESHIRE_UNDISTORT_DEVICE=1 or 0 overrides that;
// CHESHIRE_UNDISTORT_DEVICE_CHECK=1 runs both and keeps the CPU's pixels, reporting "undistort device check: N of
// M images identical" at exit.

thread_local int tlDeviceUndistort = 0;

struct CheshireRemapStats
{
    std::atomic<long long> device{0}, busy{0}, handedBack{0}, checked{0}, identical{0};
    std::atomic<bool> announced{false}, warned{false};
    ~CheshireRemapStats()
    {
        if (device.load() || busy.load() || handedBack.load())
            std::fprintf(stderr, "[cheshire] undistort on the device: %lld images (CheshireRemap), %lld on the CPU while every remapper was busy, "
                         "%lld handed back to the CPU loop\n", device.load(), busy.load(), handedBack.load());
        if (checked.load())
            std::fprintf(stderr, "[cheshire] undistort device check: %lld of %lld images identical to the CPU loop\n", identical.load(), checked.load());
    }
} g_cheshireRemapStats;

// step 9i: ExportImages' warp, counted apart
struct CheshireRemapInterStats
{
    std::atomic<long long> device{0}, busy{0}, handedBack{0}, checked{0}, identical{0};
    std::atomic<bool> announced{false}, warned{false};
    ~CheshireRemapInterStats()
    {
        if (device.load() || busy.load() || handedBack.load())
            std::fprintf(stderr, "[cheshire] export warp on the device: %lld images (CheshireRemap), %lld on the CPU while every remapper was busy, "
                         "%lld handed back to remapInter\n", device.load(), busy.load(), handedBack.load());
        if (checked.load())
            std::fprintf(stderr, "[cheshire] export warp device check: %lld of %lld images identical to remapInter\n", identical.load(), checked.load());
    }
} g_cheshireRemapInterStats;

#if defined(CHESHIRE_HAVE_REMAP)
// Remappers up to the bound, fixed at the first image from its size and the device memory then free; a thread that
// finds none idle gets nullptr and runs the CPU loop. Created once and never destroyed: at process exit the device
// runtime may already be gone.
class CheshireRemapPool
{
  public:
    // step 9i: the variable that sets the bound and the work the log line names, one pool per use
    CheshireRemapPool(const char* env, const char* what)
      : _env(env),
        _what(what)
    {}

    ::cheshire::remap::Remapper* tryAcquire(std::size_t bytesPerRemapper)
    {
        std::lock_guard<std::mutex> lock(_mutex);
        if (_max < 0)
        {
            const long long hw = std::max(1u, std::thread::hardware_concurrency());
            long long want = ::cheshire::env::integer(_env, std::min(4LL, hw));
            const long long fit = static_cast<long long>(0.6 * double(::cheshire::remap::Remapper::deviceFreeBytes()) / double(std::max<std::size_t>(1, bytesPerRemapper)));
            _max = static_cast<int>(std::max(0LL, std::min(want, fit)));
            ALICEVISION_LOG_INFO("cheshire: " << _what << " on the device: up to " << _max << " remappers (" << (bytesPerRemapper >> 20)
                                 << " MB of device memory each; " << _env << " to change)");
        }
        if (!_idle.empty())
        {
            ::cheshire::remap::Remapper* r = _idle.back();
            _idle.pop_back();
            return r;
        }
        if (static_cast<int>(_all.size()) >= _max)
            return nullptr;
        _all.emplace_back(new ::cheshire::remap::Remapper());
        return _all.back().get();
    }
    void release(::cheshire::remap::Remapper* r)
    {
        std::lock_guard<std::mutex> lock(_mutex);
        _idle.push_back(r);
    }

  private:
    const char* _env;
    const char* _what;
    std::mutex _mutex;
    std::vector<std::unique_ptr<::cheshire::remap::Remapper>> _all;
    std::vector<::cheshire::remap::Remapper*> _idle;
    int _max = -1;
};
#endif

}  // namespace

DeviceUndistortScope::DeviceUndistortScope() { ++tlDeviceUndistort; }
DeviceUndistortScope::~DeviceUndistortScope() { --tlDeviceUndistort; }
bool inDeviceUndistort() { return tlDeviceUndistort > 0; }

bool cheshireRemapDeviceEnabled()
{
#if defined(CHESHIRE_HAVE_REMAP)
    // by default on a host of four hardware threads or fewer, where the CPU loop is the costly side; on the
    // twelve-thread RX 9070 box the device's copies cost more than that CPU's loop (docs/04, 0.3.8)
    static const bool on = ::cheshire::env::flag("CHESHIRE_UNDISTORT_DEVICE", std::thread::hardware_concurrency() <= 4) &&
                           ::cheshire::remap::Remapper::deviceAvailable();
    return on;
#else
    return false;
#endif
}

bool cheshireRemapDeviceCheck()
{
    static const bool on = ::cheshire::env::flag("CHESHIRE_UNDISTORT_DEVICE_CHECK");
    return on;
}

bool cheshireRemapOnDevice(const void* key, const float* coords, int width, int height, const float* src, int srcWidth, int srcHeight,
                           const float fill[4], float* dst)
{
#if defined(CHESHIRE_HAVE_REMAP)
    if (!cheshireRemapDeviceEnabled())
        return false;
    static CheshireRemapPool* pool = new CheshireRemapPool("CHESHIRE_UNDISTORT_DEVICE_REMAPPERS", "undistort");
    const std::size_t pixels = std::size_t(width) * std::size_t(height);
    const std::size_t bytes = pixels * 2 * sizeof(float) + pixels * 4 * sizeof(float) + std::size_t(srcWidth) * std::size_t(srcHeight) * 4 * sizeof(float);
    ::cheshire::remap::Remapper* r = pool->tryAcquire(bytes);
    if (r == nullptr)
    {
        ++g_cheshireRemapStats.busy;
        return false;
    }
    ::cheshire::remap::Status st = r->setMap(key, coords, width, height);
    if (st == ::cheshire::remap::Status::Ok)
        st = r->remap(src, srcWidth, srcHeight, fill, dst);
    pool->release(r);
    if (st != ::cheshire::remap::Status::Ok)
    {
        ++g_cheshireRemapStats.handedBack;
        if (!g_cheshireRemapStats.warned.exchange(true))
            ALICEVISION_LOG_WARNING("cheshire: undistort on the device failed (" << ::cheshire::remap::statusName(st) << "); the CPU loop takes it");
        return false;
    }
    ++g_cheshireRemapStats.device;
    if (!g_cheshireRemapStats.announced.exchange(true))
        ALICEVISION_LOG_INFO("cheshire: PrepareDenseScene's undistortion on the device (CheshireRemap, AliceVision's arithmetic to the bit; "
                             "CHESHIRE_UNDISTORT_DEVICE=0 for the CPU loop, CHESHIRE_UNDISTORT_DEVICE_CHECK=1 to compare every image)");
    return true;
#else
    (void)key;
    (void)coords;
    (void)width;
    (void)height;
    (void)src;
    (void)srcWidth;
    (void)srcHeight;
    (void)fill;
    (void)dst;
    return false;
#endif
}

void cheshireRemapCheckCount(bool identical)
{
    ++g_cheshireRemapStats.checked;
    if (identical)
        ++g_cheshireRemapStats.identical;
}

// cheshire (step 9i): ExportImages' warp on the device. As for 8g, by default only on a host of four hardware threads or
// fewer: on the twelve-thread RX 9070 box the device's copies cancel what it saves the CPU (9.2 s against 9.0 s on 41
// views), with four image threads it gains 5-21 % (docs/04, 9i). CHESHIRE_EXPORT_DEVICE=1 or 0 overrides that.
bool cheshireRemapInterDeviceEnabled()
{
#if defined(CHESHIRE_HAVE_REMAP)
    static const bool on = ::cheshire::env::flag("CHESHIRE_EXPORT_DEVICE", std::thread::hardware_concurrency() <= 4) &&
                           ::cheshire::remap::Remapper::deviceAvailable();
    return on;
#else
    return false;
#endif
}

bool cheshireRemapInterDeviceCheck()
{
    static const bool on = ::cheshire::env::flag("CHESHIRE_EXPORT_DEVICE_CHECK");
    return on;
}

bool cheshireRemapInterOnDevice(const void* key, const double* map, int width, int height, const float* src, int srcWidth,
                                int srcHeight, const float fill[4], float* dst)
{
#if defined(CHESHIRE_HAVE_REMAP)
    if (!cheshireRemapInterDeviceEnabled())
        return false;
    static CheshireRemapPool* pool = new CheshireRemapPool("CHESHIRE_EXPORT_DEVICE_REMAPPERS", "export warp");
    const std::size_t pixels = std::size_t(width) * std::size_t(height);
    const std::size_t bytes = pixels * 2 * sizeof(double) + pixels * 4 * sizeof(float) + std::size_t(srcWidth) * std::size_t(srcHeight) * 4 * sizeof(float);
    ::cheshire::remap::Remapper* r = pool->tryAcquire(bytes);
    if (r == nullptr)
    {
        ++g_cheshireRemapInterStats.busy;
        return false;
    }
    ::cheshire::remap::Status st = r->setMapDouble(key, map, width, height);
    if (st == ::cheshire::remap::Status::Ok)
        st = r->remapInter(src, srcWidth, srcHeight, fill, dst);
    pool->release(r);
    if (st != ::cheshire::remap::Status::Ok)
    {
        ++g_cheshireRemapInterStats.handedBack;
        if (!g_cheshireRemapInterStats.warned.exchange(true))
            ALICEVISION_LOG_WARNING("cheshire: the export warp on the device failed (" << ::cheshire::remap::statusName(st) << "); remapInter takes it");
        return false;
    }
    ++g_cheshireRemapInterStats.device;
    if (!g_cheshireRemapInterStats.announced.exchange(true))
        ALICEVISION_LOG_INFO("cheshire: ExportImages' warp on the device (CheshireRemap, remapInter's arithmetic to the bit; "
                             "CHESHIRE_EXPORT_DEVICE=0 for the CPU, CHESHIRE_EXPORT_DEVICE_CHECK=1 to compare every image)");
    return true;
#else
    (void)key;
    (void)map;
    (void)width;
    (void)height;
    (void)src;
    (void)srcWidth;
    (void)srcHeight;
    (void)fill;
    (void)dst;
    return false;
#endif
}

void cheshireRemapInterCheckCount(bool identical)
{
    ++g_cheshireRemapInterStats.checked;
    if (identical)
        ++g_cheshireRemapInterStats.identical;
}
